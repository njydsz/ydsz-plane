### 3.1 BatchUpdateV2 批量 SQL (`coordinator_batch.go`)

```go
// 1 次查询分 3 桶 → 每桶 1~2 条批量 SQL
// 关键优化：按 type_code 分组后用 ANY($1) 批量更新
func (s *Service) BatchUpdateV2(ctx, wsID, projectID, userID, in) (BatchResult, error) {
    groups := batchGroupByType(ctx, wsID, ids)  // 1 次查询
    for tc, rows := range groups {
        batchAssigneeUpsert(tc, rows, ...)      // 每 type 1 条
        batchPriorityUpdate(tc, rows, ...)      // 每 type 1 条
    }
}
```
